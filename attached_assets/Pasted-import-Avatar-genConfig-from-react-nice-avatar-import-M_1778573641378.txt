import Avatar, { genConfig } from "react-nice-avatar";
import { Message } from "@/types/chat";
import { ChevronDown, ChevronRight, BrainCircuit, Copy, Check } from "lucide-react";
import { useState } from "react";
import ReactMarkdown from 'react-markdown';
import remarkGfm from 'remark-gfm';
import { Prism as SyntaxHighlighter } from 'react-syntax-highlighter';
import { vscDarkPlus } from 'react-syntax-highlighter/dist/esm/styles/prism';
import { toast } from "sonner";

interface MessageItemProps {
  message: Message;
}

export const MessageItem = ({ message }: MessageItemProps) => {
  const userAvatarConfig = genConfig("user@revengpt.com");
  const [isThinkingOpen, setIsThinkingOpen] = useState(false);
  const [copiedId, setCopiedId] = useState<string | null>(null);

  let content = message.content;
  let thinkingContent = null;

  // Extract all thinking tags globally (<thinking> or <think>)
  const thinkRegex = /<(thinking|think)>([\s\S]*?)(?:<\/(thinking|think)>|$)/g;
  const thinkingSteps: string[] = [];
  let match;
  
  while ((match = thinkRegex.exec(content)) !== null) {
    if (match[2]) thinkingSteps.push(match[2].trim());
  }

  if (thinkingSteps.length > 0) {
    thinkingContent = thinkingSteps.join('\n\n');
    content = content.replace(/<(thinking|think)>([\s\S]*?)(?:<\/(thinking|think)>|$)/g, "").trim();
  }

  const copyToClipboard = (text: string, id: string) => {
    navigator.clipboard.writeText(text);
    setCopiedId(id);
    toast.success("Copied to clipboard");
    setTimeout(() => setCopiedId(null), 2000);
  };

  return (
    <div className="flex gap-4 animate-in fade-in slide-in-from-bottom-2 duration-300">
      <div
        className={`h-8 w-8 rounded-full shrink-0 flex items-center justify-center overflow-hidden ${
          message.role === "user"
            ? "bg-primary text-primary-foreground shadow-sm"
            : "bg-transparent"
        }`}
      >
        {message.role === "user" ? (
          <Avatar className="h-full w-full" {...userAvatarConfig} />
        ) : (
          <img src="/logo.png" className="h-full w-full object-cover" alt="RevenGPT" />
        )}
      </div>
      <div className="flex-1 min-w-0 pt-1">
        <div className="text-sm font-semibold mb-1 opacity-80 flex items-center gap-2">
          {message.role === "user" ? "You" : "RevenGPT"}
          {message.role !== "user" && <span className="h-1.5 w-1.5 rounded-full bg-primary animate-pulse" />}
        </div>
        
        {thinkingContent && (
          <div className="mb-4 mt-2 overflow-hidden rounded-xl border border-primary/20 bg-primary/5 backdrop-blur-md max-w-[600px] transition-all hover:border-primary/30">
            <button
              onClick={() => setIsThinkingOpen(!isThinkingOpen)}
              className="flex w-full items-center gap-2 px-4 py-2.5 text-sm font-medium text-primary/80 hover:bg-primary/10 hover:text-primary transition-all group"
            >
              <BrainCircuit className="h-4 w-4 transition-transform group-hover:rotate-12" />
              <span>Thinking Process</span>
              <span className="text-xs opacity-60 font-normal">({thinkingContent.split(/\s+/).length} words)</span>
              {isThinkingOpen ? (
                <ChevronDown className="h-4 w-4 ml-auto opacity-70" />
              ) : (
                <ChevronRight className="h-4 w-4 ml-auto opacity-70" />
              )}
            </button>
            {isThinkingOpen && (
              <div className="px-4 py-3 text-[14px] text-foreground/70 whitespace-pre-wrap leading-relaxed border-t border-primary/10 bg-primary/5">
                {thinkingContent}
              </div>
            )}
          </div>
        )}

        {message.image && (
          <div className="mb-4 rounded-xl overflow-hidden border border-border inline-block max-w-[300px]">
            <img 
                src={message.image} 
                alt="Uploaded" 
                className="w-full h-auto object-cover hover:scale-105 transition-transform duration-500 cursor-pointer"
                onClick={() => window.open(message.image, '_blank')}
            />
          </div>
        )}

        <div className="text-[15px] leading-relaxed text-foreground/90 prose prose-neutral dark:prose-invert max-w-none">
          <ReactMarkdown
            remarkPlugins={[remarkGfm]}
            components={{
              code({ node, inline, className, children, ...props }: any) {
                const match = /language-(\w+)/.exec(className || '');
                const codeString = String(children).replace(/\n$/, '');
                const codeId = Math.random().toString(36).substring(7);

                return !inline && match ? (
                  <div className="relative group my-5 rounded-xl overflow-hidden border border-primary/20 bg-zinc-900/90 backdrop-blur-md shadow-2xl">
                    <div className="flex items-center justify-between px-4 py-2.5 bg-zinc-800/50 border-b border-white/5">
                      <div className="flex items-center gap-2">
                        <div className="flex gap-1.5">
                          <div className="h-2.5 w-2.5 rounded-full bg-red-500/80" />
                          <div className="h-2.5 w-2.5 rounded-full bg-amber-500/80" />
                          <div className="h-2.5 w-2.5 rounded-full bg-emerald-500/80" />
                        </div>
                        <span className="ml-2 text-[10px] font-bold text-white/40 tracking-widest uppercase">{match[1]}</span>
                      </div>
                      <button
                        onClick={() => copyToClipboard(codeString, codeId)}
                        className="flex items-center gap-1.5 px-2 py-1 rounded-md hover:bg-white/10 transition-all text-white/50 hover:text-white"
                      >
                        {copiedId === codeId ? (
                          <Check className="h-3.5 w-3.5 text-emerald-400" />
                        ) : (
                          <Copy className="h-3.5 w-3.5" />
                        )}
                        <span className="text-[11px] font-medium">{copiedId === codeId ? 'Copied' : 'Copy'}</span>
                      </button>
                    </div>
                    <SyntaxHighlighter
                      style={vscDarkPlus}
                      language={match[1]}
                      PreTag="div"
                      customStyle={{
                        margin: 0,
                        padding: '1.5rem',
                        fontSize: '13px',
                        lineHeight: '1.6',
                        background: 'transparent',
                      }}
                      {...props}
                    >
                      {codeString}
                    </SyntaxHighlighter>
                  </div>
                ) : (
                  <code className={`${className} bg-primary/10 text-primary px-1.5 py-0.5 rounded text-sm font-mono`} {...props}>
                    {children}
                  </code>
                );
              },
            }}
          >
            {content}
          </ReactMarkdown>
        </div>
      </div>
    </div>
  );
};
