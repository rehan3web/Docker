#!/bin/bash

# Docklet Installation Script for VPS
# Copyright (c) 2026 Docklet

set -e

# --- Colors for output ---
RED='\033[0;31m'
GREEN='\033[0;32m'
BLUE='\033[0;34m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

echo -e "${BLUE}==============================================${NC}"
echo -e "${BLUE}          Docklet Installation Tool           ${NC}"
echo -e "${BLUE}==============================================${NC}"

# --- Check for root ---
if [ "$EUID" -ne 0 ]; then
  echo -e "${RED}Please run as root (use sudo).${NC}"
  exit 1
fi

# --- Helper function to check if a command exists ---
command_exists() {
  command -v "$1" >/dev/null 2>&1
}

# --- Helper function to check if a port is free ---
check_port() {
  local port=$1
  if netstat -tuln | grep -q ":$port "; then
    return 1 # Port is taken
  else
    return 0 # Port is free
  fi
}

# --- 1. Check/Install Prerequisites ---
echo -e "\n${YELLOW}[1/5] Checking prerequisites...${NC}"

# Docker
if command_exists docker; then
  echo -e "${GREEN}✔ Docker is already installed.${NC}"
else
  echo -e "${BLUE}Installing Docker...${NC}"
  curl -fsSL https://get.docker.com | sh
  systemctl enable --now docker
fi

# Docker Compose
if docker compose version >/dev/null 2>&1; then
  echo -e "${GREEN}✔ Docker Compose is already installed.${NC}"
else
  echo -e "${BLUE}Installing Docker Compose...${NC}"
  # For modern Docker, it's usually a plugin
  apt-get update && apt-get install -y docker-compose-plugin || {
    # Fallback to manual binary install if needed
    DOCKER_CONFIG=${DOCKER_CONFIG:-$HOME/.docker}
    mkdir -p $DOCKER_CONFIG/cli-plugins
    curl -SL https://github.com/docker/compose/releases/download/v2.24.5/docker-compose-linux-x86_64 -o $DOCKER_CONFIG/cli-plugins/docker-compose
    chmod +x $DOCKER_CONFIG/cli-plugins/docker-compose
  }
fi

# Git
if command_exists git; then
  echo -e "${GREEN}✔ Git is already installed.${NC}"
else
  echo -e "${BLUE}Installing Git...${NC}"
  apt-get update && apt-get install -y git
fi

# --- 2. Check Port Availability ---
echo -e "\n${YELLOW}[2/5] Checking port availability...${NC}"

REQUIRED_PORTS=(80 443 3000 15432 16543)
PORT_ERROR=0

for port in "${REQUIRED_PORTS[@]}"; do
  if ! check_port $port; then
    echo -e "${RED}✘ Error: Port $port is already in use.${NC}"
    PORT_ERROR=1
  else
    echo -e "${GREEN}✔ Port $port is free.${NC}"
  fi
done

if [ $PORT_ERROR -eq 1 ]; then
  echo -e "${RED}Installation aborted. Please free the required ports and try again.${NC}"
  exit 1
fi

# --- 3. Clone Repository ---
echo -e "\n${YELLOW}[3/5] Cloning Docklet repository...${NC}"

if [ -d "docklet" ]; then
  echo -e "${YELLOW}Existing 'docklet' directory found. Updating...${NC}"
  cd docklet
  git pull
else
  git clone https://github.com/docklet-xyz/docklet.git docklet
  cd docklet
fi

# --- 4. Configuration ---
echo -e "\n${YELLOW}[4/5] Configuring environment...${NC}"

cp .env.example .env

echo -e "${BLUE}Please enter your configuration details:${NC}"

read -p "Enter your Postgres Database Name [mydb]: " DB_NAME
DB_NAME=${DB_NAME:-mydb}

read -p "Enter your Postgres User Name [myuser]: " DB_USERNAME
DB_USERNAME=${DB_USERNAME:-myuser}

read -s -p "Enter your Postgres Password: " DB_PASSWORD
echo ""

read -p "Enter your Admin Username [admin]: " ADMIN_USERNAME
ADMIN_USERNAME=${ADMIN_USERNAME:-admin}

read -s -p "Enter your Admin Password: " ADMIN_PASSWORD
echo ""

# Generate random JWT Secret
JWT_SECRET=$(openssl rand -hex 32)

# Modify .env file
# Using sed to replace the placeholders or empty values
sed -i "s/^DB_NAME=.*/DB_NAME=$DB_NAME/" .env
sed -i "s/^DB_USERNAME=.*/DB_USERNAME=$DB_USERNAME/" .env
sed -i "s/^DB_PASSWORD=.*/DB_PASSWORD=$DB_PASSWORD/" .env
sed -i "s/^ADMIN_USERNAME=.*/ADMIN_USERNAME=$ADMIN_USERNAME/" .env
sed -i "s/^ADMIN_PASSWORD=.*/ADMIN_PASSWORD=$ADMIN_PASSWORD/" .env
sed -i "s/^JWT_SECRET=.*/JWT_SECRET=$JWT_SECRET/" .env

echo -e "${GREEN}✔ .env configuration updated.${NC}"

# --- 5. Start Docklet ---
echo -e "\n${YELLOW}[5/5] Starting Docklet...${NC}"

docker compose -f postgres.yml up -d --build

echo -e "\n${GREEN}==============================================${NC}"
echo -e "${GREEN}       Docklet Installed Successfully!        ${NC}"
echo -e "${GREEN}==============================================${NC}"

# Get public IP
SERVER_IP=$(curl -s https://ifconfig.me)

echo -e "\n${BLUE}Access your Admin Panel at:${NC}"
echo -e "${YELLOW}http://$SERVER_IP:3000${NC}"
echo -e "\n${BLUE}Username: ${NC}$ADMIN_USERNAME"
echo -e "${BLUE}Password: ${NC}(The password you just entered)"

echo -e "\n${BLUE}Showing live logs (Press Ctrl+C to stop logs, Docklet will keep running):${NC}"
sleep 2
docker compose -f postgres.yml logs -f
